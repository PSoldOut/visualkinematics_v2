  28def rot_axis_from_rot_mat(rot_mat):
  29    '''
  30    Gibt die Rotationsachse von rot_mat zurueck.
  31
  32    :param rot_mat: Die Rotationsmatrix als mehrdimensionales Array. 
  33
  34    :return: Rotationsachse als normalisierter Vektor z.B [x,y,z].
  35    '''
  36    if isinstance(rot_mat,(sp.Basic, sp.MatrixBase)):
  37        rot_mat.evalf()
  38    R = np.array(rot_mat)
  39
  40    # Eigenwerte und Eigenvektoren berechnen
  41    eigenvalues, eigenvectors = np.linalg.eig(R)
  42
  43    # Eigenvektor für Eigenwert 1 finden
  44    axis = eigenvectors[:, np.isclose(eigenvalues, 1)]
  45
  46    # Ergebnis auf 1D normieren (optional)
  47    axis = axis[:, 0]
  48    axis = axis / np.linalg.norm(axis)
  49    return np.real(axis)

 141def compute_normals(vertices: np.ndarray, indices: np.ndarray) -> np.ndarray:
 142    """
 143    Berechnet Vertex-Normalen für ein Dreiecksnetz.
 144
 145    :param vertices: (N, 3) array of float32 – die Eckpunkte
 146    :param indices: (M,) array of uint32 – flach (also dreifach pro Dreieck)
 147    :return: (N, 3) array of float32 – die berechneten Vertex-Normalen
 148    """
 149    num_vertices = vertices.shape[0]
 150    normals = np.zeros((num_vertices, 3), dtype=np.float32)
 151
 152    triangles = indices.reshape(-1, 3)
 153    v0 = vertices[triangles[:, 0]]
 154    v1 = vertices[triangles[:, 1]]
 155    v2 = vertices[triangles[:, 2]]
 156
 157    # Kantenvektoren
 158    edge1 = v1 - v0
 159    edge2 = v2 - v0
 160
 161    # Flächennormalen (cross product)
 162    face_normals = np.cross(edge1, edge2)
 163
 164    # Normale zu jedem Vertex aufsummieren
 165    for i in range(3):
 166        np.add.at(normals, triangles[:, i], face_normals)
 167
 168    # Normalisieren
 169    lengths = np.linalg.norm(normals, axis=1)
 170    lengths[lengths == 0] = 1.0  # Division durch 0 vermeiden
 171    normals /= lengths[:, np.newaxis]
 172
 173    return normals

 178def order_angles(angles: list, from_order:str, to_order: str) -> list:
 179    '''
 180    Ordnet die übergebenen Eulerwinkel von einer Reihenfolge in eine andere.
 181
 182    :param angles: Liste der Winkel im Format [a1, a2, a3] in from_order.
 183    :param from_order: Der aktuelle Rotationsachsen-String (z. B. "ZYX").
 184    :param to_order: Die gewünschte neue Reihenfolge der Achsen (z. B. "XYZ").
 185    :return: Liste der Winkel in der neuen Reihenfolge to_order.
 186    '''
 187    from_order = from_order.upper()
 188    to_order = to_order.upper()
 189
 190    # Erstelle ein Mapping von Achsen zu Winkeln
 191    angle_map = {axis: angle for axis, angle in zip(from_order, angles)}
 192
 193    # Baue die neue Liste der Winkel in der gewünschten Reihenfolge
 194    return [angle_map[axis] for axis in to_order]

 199def quaternion_multiply(q1, q2):
 200    '''
 201    Multipliziert zwei Quaternions.
 202
 203    :param q1: Das erste Quaternion als Liste oder Array [x, y, z, w].
 204    :param q2: Das zweite Quaternion als Liste oder Array [x, y, z, w].
 205
 206    :return: Das Ergebnis der Quaternion-Multiplikation als Liste [x, y, z, w].
 207    '''
 208    if isinstance(q1,(sp.Basic, sp.MatrixBase)):
 209        q1.evalf()
 210    if isinstance(q2,(sp.Basic, sp.MatrixBase)):
 211        q2.evalf()
 212    x1, y1, z1, w1 = q1
 213    x2, y2, z2, w2 = q2
 214    
 215    w = w1 * w2 - x1 * x2 - y1 * y2 - z1 * z2
 216    x = w1 * x2 + x1 * w2 + y1 * z2 - z1 * y2
 217    y = w1 * y2 - x1 * z2 + y1 * w2 + z1 * x2
 218    z = w1 * z2 + x1 * y2 - y1 * x2 + z1 * w2
 219    
 220    return [x, y, z, w]

 323def create_grid_XY(size, density, pos=[0,0,0], color='#777777'):
 324    '''
 325    Erstellt ein 3D-Gitter in der XY-Ebene mit der angegebenen Größe und Dichte.
 326
 327    :param size: Die Größe des Gitters (die Ausdehnung in X- und Y-Richtung).
 328    :param density: Die Dichte des Gitters, die angibt, wie viele Linien innerhalb des Gitters erstellt werden.
 329    :param pos: Die Position des Gitters im Raum als [x, y, z]-Koordinaten.
 330    :param color: Die Linienfarbe des Gitters (Hex-String, z. B. '#777777').
 331
 332    :return: Ein 3D-Objekt (Group), das das Gitter mit Linien in der XY-Ebene enthält.
 333    '''
 334    line_material = three.LineBasicMaterial(color = color)
 335    line_material.transparent = True
 336    line_material.opacity = 0.5
 337
 338    grid_group = three.Group()
 339    for i in range((int)((-size/2)*(1/density)), (int)((size/2)*(1/density))+1):
 340        points1 = [[-size/2,i*density,0],[size/2,i*density,0]]
 341        points2 = [[i*density,-size/2,0],[i*density,size/2,0]]
 342        # Geometrie für die Linie
 343        line_geometry1 = three.BufferGeometry(
 344        attributes={'position': three.BufferAttribute(points1, False)})
 345        line1 = three.Line(line_geometry1, line_material)
 346        line_geometry2 = three.BufferGeometry(
 347        attributes={'position': three.BufferAttribute(points2, False)})
 348        line2 = three.Line(line_geometry2, line_material)
 349        grid_group.add(line1)
 350        grid_group.add(line2)
 351        grid_group.position = pos
 352    return grid_group

 390def apply_transformation_matrix(obj, transform_matrix):
 391    '''
 392    Wendet eine Transformationsmatrix auf ein 3D-Objekt an. Die Matrix bestimmt
 393    die neue Position und Orientierung (Rotation) des Objekts im Raum.
 394
 395    :param obj: Das Zielobjekt, auf das die Transformation angewendet wird. Kann ein direktes
 396                3D-Objekt oder ein Objekt mit einer `get_renderable`-Methode sein.
 397    :param transform_matrix: Eine 4x4-Transformationsmatrix (z. B. aus SymPy oder NumPy),
 398                             die Translation und Rotation des Objekts definiert.
 399
 400    :return: None. Das übergebene Objekt wird in-place transformiert.
 401    '''
 402    if isinstance(transform_matrix,(sp.Basic, sp.MatrixBase)):
 403        transform_matrix.evalf()
 404    pos = transform_matrix[:3, 3]
 405    rot = R.from_matrix(transform_matrix[:3, :3]).as_quat()
 406
 407    renderable = obj
 408    if (hasattr(obj, 'get_renderable')):
 409        renderable = obj.get_renderable()
 410    renderable.position = tuple(pos)
 411    renderable.quaternion = tuple(rot)

 415def apply_rot_matrix(obj, rot_mat):
 416    '''
 417    Wendet eine Rotationsmatrix auf ein Objekt an, indem die Matrix in ein Quaternion umgewandelt wird und auf das bestehende Quaternion des Meshs angewendet wird.
 418    Das Objekt musse entweder:
 419    - ein Attribut oder eine Methode `get_renderable()` besitzen, das ein pythreejs-kompatibles Objekt zurückgibt,
 420    - oder selbst direkt ein pythreejs-kompatibles Objekt sein (z.B. Mesh, Group, Line, etc.).
 421
 422    :param obj: Ein renderbares Objekt oder ein Wrapper mit `get_renderable()`, auf das die Rotation angewendet werden soll. Erwartet wird, dass das Mesh ein `quaternion`-Attribut besitzt.
 423    :param rot_mat: Die Rotationsmatrix, die auf das Mesh angewendet werden soll. Muss eine 3x3 Matrix sein.
 424    '''
 425    if isinstance(rot_mat,(sp.Basic, sp.MatrixBase)):
 426        rot_mat.evalf()
 427    # Konvertiere Matrix in Quaternion
 428    r = R.from_matrix(rot_mat)
 429    q = r.as_quat()  # Reihenfolge: [x, y, z, w]
 430
 431    renderable = obj
 432    if (hasattr(obj, 'get_renderable')):
 433        renderable = obj.get_renderable()
 434
 435    # Setze Quaternion (pythreejs erwartet [w, x, y, z])
 436    renderable.quaternion = quaternion_multiply((q[0], q[1], q[2], q[3]), renderable.quaternion)

 443def create_colored_quad(pos, width, height, depth,
 444                        face_colors=[[255, 0, 0], [0, 255, 0], [0, 0, 255], [255, 255, 0], [255, 0, 255], [0, 255, 255]],
 445                        transparent=True):
 446    '''
 447    Erzeugt ein Quader-Mesh mit individuell gefärbten Seiten.
 448

 450    :param width: Breite des Quaders

 452    :param depth: Tiefe des Quaders
 453    :param face_colors: Liste von 6 RGB-Farben, eine pro Seite. Reihenfolge:

 471
 472    mesh = three.Mesh(geometry=geometry, material=materials)
 473    mesh.position = tuple(pos)

 479def create_box(pos, width, height, depth, color=[0,255,0], transparent=True):
 480    '''
 481    Erzeugt ein Quader-Mesh (Box) mit der angegebenen Position, Größe und Farbe.
 482
 483    :param pos: Die Position des Quaders als Array oder Tuple [x, y, z].
 484    :param width: Die Breite des Quaders.
 485    :param height: Die Höhe des Quaders.

 598def slerp_quaternion(q1, q2, t):
 599    '''
 600    Führt eine Spherical Linear Interpolation (SLERP) zwischen zwei Quaternionen durch.
 601    Interpoliert die Rotation zwischen q1 und q2 basierend auf dem Interpolationswert t.
 602
 603    :param q1: Das erste Quaternion, das die Anfangsrotation beschreibt.
 604    :param q2: Das zweite Quaternion, das die Endrotation beschreibt.
 605    :param t: Der Interpolationswert, der zwischen 0 und 1 liegen muss. Ein Wert von 0 entspricht der Rotation von q1 und ein Wert von 1 entspricht der Rotation von q2.
 606
 607    :return: Das interpolierte Quaternion, das die Rotation zwischen q1 und q2 bei dem gegebenen Wert von t beschreibt.
 608    
 609    :raises ValueError: Wenn der Interpolationswert t nicht zwischen 0 und 1 liegt.
 610    '''
 611
 612    if isinstance(q1,(sp.Basic, sp.MatrixBase)):
 613        q1.evalf()
 614    if isinstance(q2,(sp.Basic, sp.MatrixBase)):
 615        q2.evalf()
 616    if not (0.0 <= t <= 1.0):
 617        raise ValueError(f"Der Interpolationswert t muss zwischen 0 und 1 liegen. t ist aber {t}")
 618    
 619
 620    # Erstelle Rotationsobjekte
 621    key_times = np.array([0, 1])  # Start (0) und Ende (1)
 622    key_rots = R.from_quat([q1, q2])  # Quaternionen als Rotation-Objekte
 623
 624    # SLERP-Interpolation erstellen
 625    slerp = Slerp(key_times, key_rots)
 626
 627    # Interpolierte Rotation abrufen
 628    interpolated_rotation = slerp(t)
 629
 630    return interpolated_rotation.as_quat()

 846def move(robot, vel, steps=1):
 847    '''
 848    Bewegt ein Roboterobjekt in mehreren Schritten entsprechend der gegebenen Geschwindigkeit und Rotation.
 849
 850    Die Funktion kombiniert Translation und Rotation:
 851    - Zuerst wird eine Rotation um die Z-Achse angewendet, basierend auf dem dritten Element des Geschwindigkeitsvektors `vel[2]`.
 852    - Anschließend wird eine Translation basierend auf der aktuellen Ausrichtung (Z-Rotation) des Roboters ausgeführt.
 853    - Die Bewegung wird für eine angegebene Anzahl von Schritten (`steps`) wiederholt.
 854    - Nach jeweils 4 Schritten erfolgt eine kurze Pause zur visuellen Glättung.
 855
 856    :param robot: Ein renderbares Objekt oder ein Wrapper mit `get_renderable()`, das bewegt werden soll.
 857    :param vel: Ein Geschwindigkeitsvektor `[v_x, v_y, ω_z]`, wobei `v_x` und `v_y` die Translation in der lokalen X- und Y-Richtung und `ω_z` die Rotation um die Z-Achse ist (in Radiant).
 858    :param steps: Die Anzahl der Schritte, die die Bewegung ausführen soll (Standard: 1).
 859
 860    :return: Die finale Position des Roboters nach der Bewegung.
 861    '''
 862    renderable = robot
 863    if (hasattr(robot, 'get_renderable')):
 864        renderable = robot.get_renderable()
 865
 866    if isinstance(vel,(sp.Basic, sp.MatrixBase)):
 867        vel.evalf()
 868    for i in range(steps):
 869        rot_mat_z = np.array([
 870        [np.cos(vel[2]), -np.sin(vel[2]), 0],
 871        [np.sin(vel[2]),  np.cos(vel[2]), 0],
 872        [0,             0,             1]
 873        ])
 874
 875        apply_rot_matrix(renderable, rot_mat_z)
 876        x = renderable.quaternion[0]
 877        y = renderable.quaternion[1]
 878        z = renderable.quaternion[2]
 879        w = renderable.quaternion[3]
 880
 881        z_angle = quaternion_to_euler(x,y,z,w,"XYZ")[2]
 882        cos_z = np.cos(np.radians(z_angle))
 883        sin_z = np.sin(np.radians(z_angle))
 884        translate(renderable, [cos_z*vel[0] + sin_z*vel[1], sin_z*vel[0] + cos_z*vel[1], 0])
 885        if (i%4==0 and i!=0):
 886            time.sleep(0.01)
 887    return renderable.position

143    """
144    Berechnet Vertex-Normalen für ein Dreiecksnetz.
145
146    :param vertices: (N, 3) array of float32 – die Eckpunkte
147    :param indices: (M,) array of uint32 – flach (also dreifach pro Dreieck)

156    v2 = vertices[triangles[:, 2]]
157

171    lengths[lengths == 0] = 1.0  # Division durch 0 vermeiden
